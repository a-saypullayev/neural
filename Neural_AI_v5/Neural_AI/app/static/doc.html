Getting started

Neural AI is a private chat app with an OpenAI-style REST API. To call the API you need an account and an API key.

  1. Create an account with a username and password.
  2. Open API keys and generate a key.
  3. Send requests to {{BASE}}/v1/chat/completions with your key.

Creating an account

Choose Create account on the sign-in screen and pick a username (3-30 letters, numbers, dot, dash or underscore) and a password of at least 8 characters. No email is needed. Sign in later with the same username and password.

API key generation

On the API keys page, name your key and press Generate key. The full key (it starts with nrl_) is shown once: copy it immediately. Only a SHA-256 hash is stored, so we cannot show it again. You can see each key's name, prefix, creation date and last use, and revoke it at any time. Revocation takes effect immediately. On the Free plan you can hold up to 3 active keys and create up to 3 new keys per day (UTC). Each key has its own allowance of 10,000 tokens per day.

Authentication

Send your key as a bearer token on every /v1 request:

Authorization: Bearer nrl_your_key_here

Keep keys secret. Do not put them in public web pages or commit them to Git; call the API from your server and store the key in an environment variable.

Endpoints

GET /v1/models

Lists available models. The model id is neural-1.

curl {{BASE}}/v1/models -H "Authorization: Bearer $NEURAL_API_KEY"

POST /v1/chat/completions

FieldTypeDescription
modelstringOptional. neural-1.
messagesarrayRequired. Objects with role (system, user, assistant) and text content. The last message must be from user. Max 50 messages, 32,000 characters in total.
streambooleanSet true for server-sent events.
temperaturenumber0 to 2.
max_tokensinteger1 to 2048 (default 2048).

Response (non-streaming):

{
  "id": "chatcmpl-...", "object": "chat.completion", "created": 1767225600, "model": "neural-1",
  "choices": [{"index": 0, "message": {"role": "assistant", "content": "Hello!"}, "finish_reason": "stop"}],
  "usage": {"prompt_tokens": 12, "completion_tokens": 3, "total_tokens": 15}
}

Replies are generated live by the Neural AI model. usage is reported by the model server and may be null.

Examples

Set NEURAL_API_KEY (and optionally NEURAL_BASE_URL) in your environment, then run:

"""Call the Neural API from plain Python (pip install requests)."""
import json
import os

import requests

BASE_URL = os.environ.get("NEURAL_BASE_URL", "{{BASE}}")
API_KEY = os.environ["NEURAL_API_KEY"]
HEADERS = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}

# 1) Non-streaming
r = requests.post(f"{BASE_URL}/v1/chat/completions", headers=HEADERS, timeout=120, json={
    "model": "neural-1",
    "messages": [{"role": "user", "content": "Say hello in five words."}],
})
r.raise_for_status()
print(r.json()["choices"][0]["message"]["content"])

# 2) Streaming (server-sent events)
with requests.post(f"{BASE_URL}/v1/chat/completions", headers=HEADERS, stream=True, timeout=120, json={
    "model": "neural-1",
    "stream": True,
    "messages": [{"role": "user", "content": "Count to five."}],
}) as resp:
    resp.raise_for_status()
    for line in resp.iter_lines(decode_unicode=True):
        if not line or not line.startswith("data: "):
            continue
        data = line[6:]
        if data == "[DONE]":
            break
        delta = json.loads(data)["choices"][0]["delta"].get("content", "")
        print(delta, end="", flush=True)
print()

// Works in Node 18+ and in browsers. Never put your API key in public browser code:
// call the API from your own server instead.
const BASE_URL = process.env.NEURAL_BASE_URL || "{{BASE}}";
const API_KEY = process.env.NEURAL_API_KEY;

async function main() {
  const res = await fetch(`${BASE_URL}/v1/chat/completions`, {
    method: "POST",
    headers: { Authorization: `Bearer ${API_KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "neural-1",
      messages: [{ role: "user", content: "Say hello in five words." }],
    }),
  });
  const data = await res.json();
  if (!res.ok) throw new Error(`${res.status}: ${data.error.message}`);
  console.log(data.choices[0].message.content);
}
main().catch((e) => { console.error(e.message); process.exit(1); });

// Node.js 18+ streaming example (no dependencies).
const BASE_URL = process.env.NEURAL_BASE_URL || "{{BASE}}";
const API_KEY = process.env.NEURAL_API_KEY;

async function main() {
  const res = await fetch(`${BASE_URL}/v1/chat/completions`, {
    method: "POST",
    headers: { Authorization: `Bearer ${API_KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify({ model: "neural-1", stream: true, messages: [{ role: "user", content: "Count to five." }] }),
  });
  if (!res.ok) throw new Error(`${res.status}: ${(await res.json()).error.message}`);
  const decoder = new TextDecoder();
  let buffer = "";
  for await (const chunk of res.body) {
    buffer += decoder.decode(chunk, { stream: true });
    let i;
    while ((i = buffer.indexOf("\n\n")) >= 0) {
      const event = buffer.slice(0, i).trim();
      buffer = buffer.slice(i + 2);
      if (!event.startsWith("data: ")) continue;
      const data = event.slice(6);
      if (data === "[DONE]") { process.stdout.write("\n"); return; }
      const delta = JSON.parse(data).choices?.[0]?.delta?.content;
      if (delta) process.stdout.write(delta);
    }
  }
}
main().catch((e) => { console.error(e.message); process.exit(1); });

$status = curl_getinfo($ch, CURLINFO_HTTP_CODE);
curl_close($ch);

$data = json_decode($body, true);
if ($status !== 200) {
    fwrite(STDERR, "Error $status: " . ($data["error"]["message"] ?? "unknown") . "\n");

#!/usr/bin/env sh
# Usage: NEURAL_API_KEY=nrl_... NEURAL_BASE_URL=https://your-app.onrender.com sh curl.sh
BASE="${NEURAL_BASE_URL:-{{BASE}}}"

echo "== Models"
curl -s "$BASE/v1/models" -H "Authorization: Bearer $NEURAL_API_KEY"

echo; echo "== Chat completion"
curl -s "$BASE/v1/chat/completions" \
  -H "Authorization: Bearer $NEURAL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"model":"neural-1","messages":[{"role":"user","content":"Say hello in five words."}]}'

echo; echo "== Streaming (-N disables buffering)"
curl -N -s "$BASE/v1/chat/completions" \
  -H "Authorization: Bearer $NEURAL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"model":"neural-1","stream":true,"messages":[{"role":"user","content":"Count to five."}]}'

Streaming

With "stream": true the response is text/event-stream. Each event is data: {...} with an object of type chat.completion.chunk; the text is in choices[0].delta.content. The stream ends with data: [DONE]. If the model fails mid-stream, an event containing an error object is sent before the end.

Errors

Errors use this shape: {"error": {"message": "...", "type": "...", "code": "..."}}

StatusCodeMeaning
400invalid_request, invalid_messages, input_too_long ...The request body is wrong or too large.
401missing_api_key, invalid_api_keyNo key, or the key is invalid, revoked, or has been revoked.
404model_not_foundUnknown model.
429rate_limit_exceeded, daily_limitToo many requests. Honor the Retry-After header.
502 / 503upstream_error, model_unavailableThe AI model is unreachable or failed. Retry shortly.

Rate limits

  • 30 requests per minute per API key (headers X-RateLimit-Limit and X-RateLimit-Remaining; Retry-After on 429).
  • 10,000 tokens per day, per API key on the Free plan (resets 00:00 UTC). Tokens are input plus output; the response usage field shows what each call used, and the headers X-Tokens-Limit and X-Tokens-Remaining show the key's budget. When a key is out of tokens the API returns HTTP 429 with code daily_limit.
  • 3 active API keys and 3 new keys per day per account on the Free plan.
  • 500 API requests per day per account (resets 00:00 UTC).
  • Operators can change these limits through environment variables.

Python SDK

The neuralai package lives in the sdk/ folder of this project. It is not published on PyPI yet, so install it from the folder:

pip install ./sdk

Once you publish it to PyPI, pip install neuralai will work.

Chat

"""Basic chat completion. Set NEURAL_API_KEY and NEURAL_BASE_URL first."""
from neuralai import Neural

client = Neural()  # reads NEURAL_API_KEY and NEURAL_BASE_URL from the environment

print([m.id for m in client.models.list().data])

reply = client.chat.completions.create(
    model="neural-1",
    messages=[
        {"role": "system", "content": "You are concise."},
        {"role": "user", "content": "Explain what an API key is in one sentence."},
    ],
    temperature=0.5,
    max_tokens=200,
)
print(reply.choices[0].message.content)
print(reply.usage)

Streaming

"""Stream a reply token by token."""
from neuralai import Neural

client = Neural()
with client.chat.completions.create(messages=[{"role": "user", "content": "Write a haiku about the sea."}], stream=True) as stream:
    for text in stream.iter_text():
        print(text, end="", flush=True)
print()

Error handling

"""Handle the errors the API can return."""
import time

from neuralai import (APIConnectionError, AuthenticationError, InvalidRequestError, Neural, RateLimitError, ServerError)

client = Neural()
try:
    r = client.chat.completions.create(messages=[{"role": "user", "content": "Hello!"}])
    print(r.choices[0].message.content)
except AuthenticationError:
    print("Your API key is invalid or revoked. Create a new one at /keys.")
except RateLimitError as e:
    print(f"Slow down. Retry in {e.retry_after or 60}s ({e.code}).")
    time.sleep(e.retry_after or 60)
except InvalidRequestError as e:
    print("Bad request:", e.message)
except ServerError as e:
    print("The model is unavailable right now:", e.message)
except APIConnectionError as e:
    print("Could not reach the server:", e)
